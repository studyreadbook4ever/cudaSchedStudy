cmake_minimum_required(VERSION 3.24)
project(gpu_host LANGUAGES CXX CUDA)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CUDA_STANDARD 17)
if(NOT DEFINED CMAKE_CUDA_ARCHITECTURES)
  set(CMAKE_CUDA_ARCHITECTURES 75 CACHE STRING "CUDA targets; override for your GPU")
endif()
find_package(CUDAToolkit 13.1 REQUIRED)
add_library(gpu_backend src/gpu_backend.cu)
target_include_directories(gpu_backend PUBLIC include)
target_link_libraries(gpu_backend PUBLIC CUDA::cudart CUDA::cuda_driver)
target_compile_options(gpu_backend PRIVATE $<$<COMPILE_LANGUAGE:CUDA>:-lineinfo>)
add_executable(gpu-host src/server.cpp)
target_link_libraries(gpu-host PRIVATE gpu_backend)
target_compile_options(gpu-host PRIVATE -Wall -Wextra -Wpedantic -Wconversion -Wshadow)
add_executable(gpu_probe tools/gpu_probe.cu)
target_link_libraries(gpu_probe PRIVATE CUDA::cudart CUDA::cuda_driver)
target_compile_options(gpu_probe PRIVATE -lineinfo)
if(EXISTS "${CMAKE_SOURCE_DIR}/tools/backend_smoke.cu")
  add_executable(backend_smoke tools/backend_smoke.cu)
  target_link_libraries(backend_smoke PRIVATE gpu_backend)
  target_compile_options(backend_smoke PRIVATE -lineinfo)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/tools/backend_stress.cu")
  add_executable(backend_stress tools/backend_stress.cu)
  target_link_libraries(backend_stress PRIVATE gpu_backend)
  target_compile_options(backend_stress PRIVATE -lineinfo)
endif()
if(EXISTS "${CMAKE_SOURCE_DIR}/tools/backend_bench.cu")
  add_executable(backend_bench tools/backend_bench.cu)
  target_link_libraries(backend_bench PRIVATE gpu_backend)
  target_compile_options(backend_bench PRIVATE -lineinfo)
endif()
enable_testing()
find_package(Python3 COMPONENTS Interpreter REQUIRED)
add_test(NAME integration COMMAND ${Python3_EXECUTABLE} -m unittest discover -s tests -v)
set_tests_properties(integration PROPERTIES WORKING_DIRECTORY "${CMAKE_SOURCE_DIR}" TIMEOUT 180
  ENVIRONMENT "GPU_HOST_BINARY=$<TARGET_FILE:gpu-host>")
